def validate_redirect(status,headers,location):
    u=urlsplit(location)
    if status not in [302,303,307,308]:raise RuntimeError('Fresh publisher redirect required')
    if u.scheme!='https' or u.hostname!=HOST or u.port not in [None,443] or u.username is not None or u.password is not None or u.fragment:raise RuntimeError('Exact approved HTTPS CDN identity mismatch')
    if not u.path.startswith('/xet-bridge-us/') or XET not in u.path:raise RuntimeError('Pinned Xet object path mismatch')
    if headers.get('x-repo-commit')!=REV or headers.get('x-linked-size')!=str(SIZE) or headers.get('x-linked-etag','').strip('"')!=SHA or headers.get('x-xet-hash')!=XET:raise RuntimeError('Publisher security/identity headers mismatch')
    return u

def main(root):
    start=time.monotonic();received=0;connections=[]
    def log(kind,**data):
        with (root/'download-progress.jsonl').open('a') as f:
            f.write(json.dumps({'event':kind,'elapsed_s':time.monotonic()-start,**data})+'\n');f.flush();os.fsync(f.fileno())
    def connection(host):
        remaining=720-(time.monotonic()-start)
        if remaining<=0:raise TimeoutError('Source-specific model network deadline')
        context=ssl.create_default_context()
        conn=http.client.HTTPSConnection(host,443,context=context,timeout=min(30,remaining));connections.append(conn)
        return conn
    receipt={'task':'N15-04B','canonical_url':'https://huggingface.co'+PATH,'revision':REV,'artifact':FILE,'expected_bytes':SIZE,'expected_sha256':SHA,'complete':False,'attempts':1,'redirects':[],'started_utc':datetime.datetime.now(datetime.timezone.utc).isoformat(),'signed_query_values_retained':False,'new_runtime_bytes':0}
    try:
        log('publisher_request_started',host='huggingface.co')
        publisher=connection('huggingface.co')
        publisher.request('GET',PATH,headers={'User-Agent':'ASTRA-N15-04B-bounded-acquisition/1.0','Accept':'application/octet-stream','Accept-Encoding':'identity','Connection':'close'})
        response=publisher.getresponse()
        if sum(len(k)+len(v) for k,v in response.getheaders())>65536:raise RuntimeError('Publisher metadata header bound')
        location=response.getheader('Location')
        diag=sanitized_redirect(location,response.status)
        headers={k.lower():v for k,v in response.getheaders()}
        try:u=validate_redirect(response.status,headers,location)
        except Exception as exc:
            log('redirect_rejected',**sanitized_redirect(location,response.status,exc));raise
        receipt['publisher']={'status':response.status,'TLS_certificate_hostname_verification':True,'x_repo_commit':headers['x-repo-commit'],'x_linked_size':int(headers['x-linked-size']),'x_linked_etag':headers['x-linked-etag'].strip('"'),'x_xet_hash':headers['x-xet-hash']}
        receipt['redirects'].append({**diag,'path_prefix':'/xet-bridge-us/','expected_object_marker_matches':True})
        log('publisher_bindings_verified',hostname=u.hostname,status=response.status)
        response.close();publisher.close()
        cdn=connection(HOST)
        cdn.request('GET',u.path+'?'+u.query,headers={'User-Agent':'ASTRA-N15-04B-bounded-acquisition/1.0','Accept':'application/octet-stream','Accept-Encoding':'identity','Connection':'close'})
        response=cdn.getresponse()
        if response.status!=200:
            if response.status in [301,302,303,307,308]:log('unexpected_extra_redirect',**sanitized_redirect(response.getheader('Location'),response.status))
            raise RuntimeError('CDN response not complete HTTP200; no follow/retry')
        if response.getheader('Content-Range') or response.getheader('Content-Encoding','identity')!='identity' or response.getheader('Content-Length')!=str(SIZE):raise RuntimeError('CDN size/range/encoding mismatch')
        mime=response.getheader('Content-Type','').split(';')[0].strip()
        if mime not in ['application/octet-stream','binary/octet-stream','application/x-binary','application/x-gguf']:raise RuntimeError('CDN model content type mismatch')
        receipt['delivery']={'host':HOST,'HTTPS_port':443,'TLS_certificate_hostname_verification':True,'status':200,'content_length':SIZE,'content_type':mime}
        digest=hashlib.sha256();last=0
        with (root/(FILE+'.part')).open('xb') as output:
            while True:
                if time.monotonic()-start>=720:raise TimeoutError('Source-specific model network deadline')
                chunk=response.read1(min(1024*1024,SIZE+1-received))
                if not chunk:break
                received+=len(chunk)
                if received>SIZE or received>550000000:raise RuntimeError('Model size/acquisition budget exceeded')
                output.write(chunk);digest.update(chunk)
                if time.monotonic()-last>=1:
                    output.flush();os.fsync(output.fileno());log('receiving',bytes=received);last=time.monotonic()
            output.flush();os.fsync(output.fileno())
        if received!=SIZE or digest.hexdigest()!=SHA:raise RuntimeError('Complete original length/SHA256 mismatch')
        receipt.update(complete=True,received_bytes=received,sha256=digest.hexdigest())
        log('complete_original_received',bytes=received,sha256=SHA)
    except BaseException as exc:
        reason=str(exc) if str(exc) in SAFE_ERRORS else ('TLS certificate verification failed' if isinstance(exc,ssl.SSLCertVerificationError) else 'Acquisition failed; untrusted exception text omitted.')
        receipt['failure']={'type':type(exc).__name__,'reason':reason}
        log('failed',error_type=type(exc).__name__,reason=reason,received_bytes=received)
    finally:
        for conn in connections:conn.close()
        receipt.update(received_bytes=received,elapsed_s=time.monotonic()-start,support_redirect_body_bytes=0,total_received_body_bytes=received,CDN_permission_active=False,CDN_permission_revoked='One-shot acquisition worker closed all connections and exits; no global network rule installed.')
        with (root/'download-receipt.json').open('x') as f:json.dump(receipt,f,indent=2);f.write('\n');f.flush();os.fsync(f.fileno())
    if not receipt['complete']:raise SystemExit(1)

if __name__=='__main__':main(Path(sys.argv[1]))
