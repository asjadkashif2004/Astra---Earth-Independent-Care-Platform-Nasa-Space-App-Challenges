def scan():

def trusted(args,name,timeout=20):
    gate()
    with (root/(name+'.stdout')).open('wb') as out,(root/(name+'.stderr')).open('wb') as err:
        with active_lock:
            if halt.is_set() or failures:raise RuntimeError('Closed resource gate')
            p=subprocess.Popen(args,stdin=subprocess.DEVNULL,stdout=out,stderr=err,env=env(),close_fds=True,preexec_fn=child_limits)
            active[p.pid]=p
        try:code=p.wait(timeout=timeout)
        except BaseException:p.kill();p.wait();raise
        finally:
            with active_lock:active.pop(p.pid,None)
    gate()
    return {'exit_code':code,'stdout':(root/(name+'.stdout')).read_text(errors='replace'),'stderr':(root/(name+'.stderr')).read_text(errors='replace')}
