def digest(path):
    h=hashlib.sha256()
    with path.open('rb') as stream:
        for block in iter(lambda:stream.read(1024*1024),b''):h.update(block)
    return h.hexdigest()

for group in ['source','history']:
    for name,expected in before[group].items():
        if group=='source' and name in changed:continue
        s=(p/name).stat()
        assert {'size':s.st_size,'mtime_ns':s.st_mtime_ns}==expected,(group,name)
for name,expected in before['pins'].items():
    if name not in changed:assert digest(p/name)==expected,name

for name,expected in before['external'].items():
    path=Path(name);s=path.stat()
    actual={'size':s.st_size,'mtime_ns':s.st_mtime_ns,'sha256':digest(path)}
    assert actual==expected,name
    external[name]=actual

for name in ['load_frozen','validate_layout','validate_sandbox_globals','validate_grammar_request']:
    find=lambda tree:next(n for n in tree.body if isinstance(n,ast.FunctionDef) and n.name==name)
    assert ast.dump(find(old_tree))==ast.dump(find(new_tree)),name
for name in ['PINS','POLICY','FILE','SHA','B_BASE','prefix','options']:
    def assignment(tree):
        return next(n for n in ast.walk(tree) if isinstance(n,ast.Assign)
                    and any(isinstance(t,ast.Name) and t.id==name for t in n.targets))
    assert ast.dump(assignment(old_tree))==ast.dump(assignment(new_tree)),name
