import { CREW, METRICS, MISSION_ID, validateTelemetry } from '../../packages/contracts.mjs';
import { METRIC_NAMES, exact, fail, sha, stable } from './read-contract.mjs';

export const SQLITE_EDITION='astra.n16.scratch-sqlite.v1';
export const AS_OF='2026-11-14T11:00:00.000Z';
export const PROFILES=Object.freeze(['nominal','elevated','missing','stale','malformed','untrusted']);
export const CAPS=Object.freeze({db:1048576,wal:4194304,storage:67108864,rss:536870912,toolMs:1000,runMs:10000,transactionMs:250,cleanupMs:250,result:65536});
export const SCHEMA=`CREATE TABLE fixture_meta(key TEXT PRIMARY KEY,json TEXT NOT NULL);
CREATE TABLE observations(crew_id TEXT NOT NULL,metric TEXT NOT NULL,json TEXT NOT NULL,sha256 TEXT NOT NULL,PRIMARY KEY(crew_id,metric));
CREATE TABLE baselines(crew_id TEXT NOT NULL,metric TEXT NOT NULL,json TEXT NOT NULL,sha256 TEXT NOT NULL,PRIMARY KEY(crew_id,metric));
CREATE TABLE risks(id TEXT PRIMARY KEY,crew_id TEXT NOT NULL,status TEXT NOT NULL,json TEXT NOT NULL,sha256 TEXT NOT NULL);
CREATE INDEX risk_scope ON risks(crew_id,status,id);`;
export const QUERIES=Object.freeze({
  metadata:'SELECT key,json FROM fixture_meta WHERE key IN (?,?,?) ORDER BY key',
  observation:'SELECT crew_id,metric,json,sha256 FROM observations WHERE crew_id=? AND metric=?',
  baseline:'SELECT crew_id,metric,json,sha256 FROM baselines WHERE crew_id=? AND metric=?',
  risks:'SELECT id,crew_id,status,json,sha256 FROM risks WHERE crew_id=? AND status=? ORDER BY id LIMIT ?'
});
export const WRITER_PLAN=Object.freeze({edition:SQLITE_EDITION,transition:'generation 0 -> 1; CREW-03 heart_rate 103 -> 104',
  commands:['commit','checkpoint','truncate','exit'],test_only:true,N02_executed:false});
export function rowsFor(profile,generation=0) {
  if(!PROFILES.includes(profile)||![0,1].includes(generation))fail('PROFILE_FORBIDDEN');
  const rows={observations:[],baselines:[],risks:[]};
  for(const [i,c] of CREW.entries()) for(const m of METRIC_NAMES){
    let value=m==='heart_rate'?68+i:m==='respiratory_rate'?14:98.1;
    if(profile==='elevated'&&c.id==='CREW-03'&&m==='heart_rate')value=generation?104:103;
    const target=c.id==='CREW-03'&&m==='heart_rate';
    const ts=profile==='stale'&&target?'2026-11-14T10:59:40.000Z':AS_OF;
    const def=METRICS[m],e={schema_version:'astra.telemetry.v1',event_id:`N16SQL:${profile}:${c.id}:${m}:g${profile==='elevated'&&target?generation:0}`,
      mission_id:MISSION_ID,crew_id:c.id,ts,received_ts:ts,
      metric:{system:def.system,code:def.code,name:m,unit:def.unit,value:profile==='malformed'&&target?'not-a-number':value},
      quality:{status:'good',confidence:0.99,artifact:false},
      source:{kind:profile==='untrusted'&&target?'synthetic-import':'simulator',device_id:`SQL-FIXTURE-${c.id}`,sampling_hz:1,runtime_hz:1,clock:'mission_simulation'},
      provenance:{synthetic:true,scenario_id:`n16-sql-${profile}`,seed:42017,generator_version:SQLITE_EDITION}};
    if(!(profile==='missing'&&target))rows.observations.push({crew_id:c.id,metric:m,json:stable(e),sha256:sha(stable(e))});
    const b={center:def.center,scale:def.scale,unit:def.unit,method:'authored_fixture_reference',synthetic:true,count:null,window_days:null,version:SQLITE_EDITION};
    rows.baselines.push({crew_id:c.id,metric:m,json:stable(b),sha256:sha(stable(b))});
  }
  const r={id:'N16SQL-RISK-SCHEMA-01',crew_id:'CREW-03',status:'open',record_origin:'authored_synthetic_fixture',synthetic:true,
    actual_N02_event:false,diagnosis:null,requires_human_acknowledgement:true,observed_at:AS_OF,
    description:'Authored persisted schema example; no detector or alert creation was executed.'};
  rows.risks.push({id:r.id,crew_id:r.crew_id,status:r.status,json:stable(r),sha256:sha(stable(r))});return rows;
}
export function sourceMeta(profile,generation){return {schema_version:SQLITE_EDITION,profile,generation};}
export function verifyRow(row,expected) {
  if(!row||!expected||!exact(row,Object.keys(expected))||stable(row)!==stable(expected)||sha(row.json)!==row.sha256)fail('ROW_PROVENANCE_MISMATCH');
  return JSON.parse(row.json);
}
export function projectRows(profile,generation,call,selected) {
  const expected=rowsFor(profile,generation),p={source_kind:'scratch_sqlite',fixture_id:profile,edition:SQLITE_EDITION,generation,
    as_of:AS_OF,clock:'mission_simulation',synthetic:true,mission_id:MISSION_ID,earth_received_at:null,host_currency:'NOT_LIVE_FIXTURE'};
  const base={schema_version:'astra.n16.sqlite-result.v1',tool:call.tool,crew_id:call.args.crew_id,synthetic:true,research_only:true,
    inference_performed:false,clinical_support:false,provenance:p};
  if(call.tool==='get_open_risk_events'){
    const want=expected.risks.filter(r=>r.crew_id===call.args.crew_id&&r.status==='open').slice(0,call.args.limit+1);
    if(stable(selected)!==stable(want))fail('ROW_PROVENANCE_MISMATCH');
    const records=selected.slice(0,call.args.limit).map((r,i)=>verifyRow(r,want[i]));
    return {...base,data_status:records.length?'KNOWN':'UNKNOWN',data:{records,coverage:'authored_fixture_only',
      N02_provenance:'UNKNOWN',actual_N02_event:false,truncated:selected.length>call.args.limit,
      reason:'Only synthetic schema records; no N02 output or clinical all-clear is established.'}};
  }
  const table=call.tool==='get_personal_baseline'?'baselines':'observations';
  const items=call.args.metrics.map((m,i)=>{
    const wanted=expected[table].find(r=>r.crew_id===call.args.crew_id&&r.metric===m),row=selected[i];
    if(!wanted){if(row!==null)fail('UNEXPECTED_ROW');return {metric:m,unit:METRICS[m].unit,value:null,data_status:'UNKNOWN',reason:'missing_observation',provenance:null};}
    const value=verifyRow(row,wanted);
    if(table==='baselines')return {metric:m,unit:METRICS[m].unit,value,data_status:'KNOWN',reason:'authored_reference_not_personal_archive',provenance:{row_sha256:row.sha256}};
    let reason=null,age=Date.parse(AS_OF)-Date.parse(value.ts);
    try{validateTelemetry(value,Date.parse(AS_OF));}catch{reason='invalid_observation';}
    if(!reason&&value.source.kind!=='simulator')reason='source_not_authorized';
    if(!reason&&age>Math.max(5000,3000/value.source.sampling_hz))reason='stale_observation';
    if(!reason&&(value.quality.status!=='good'||value.quality.artifact||value.metric.value===null))reason='unreliable_quality';
    return {metric:m,unit:METRICS[m].unit,value:!reason&&call.tool==='get_current_vitals'?value.metric.value:null,
      data_status:reason?'UNKNOWN':'KNOWN',reason,quality:value.quality,
      provenance:{event_id:value.event_id,row_sha256:row.sha256,observed_at:value.ts,received_at:value.received_ts,
        source:value.source,source_provenance:value.provenance,age_ms:age,freshness:reason==='stale_observation'?'stale':reason?'unverified':'fresh'}};
  });
  const known=items.filter(x=>x.data_status==='KNOWN').length;
  return {...base,data_status:known===items.length?'KNOWN':known?'PARTIAL':'UNKNOWN',data:{items}};
}
