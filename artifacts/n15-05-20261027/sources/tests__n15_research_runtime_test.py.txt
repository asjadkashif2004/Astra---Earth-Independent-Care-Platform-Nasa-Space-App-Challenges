"""Static-only runner controls. Never creates a sandbox or loads acquired code."""
import ast,copy,importlib.util,tempfile,unittest
from pathlib import Path

path=Path(__file__).resolve().parents[1]/'scripts/n15_research_runtime.py'
spec=importlib.util.spec_from_file_location('research_runtime',path)
runner=importlib.util.module_from_spec(spec);spec.loader.exec_module(runner)

class RuntimeStaticTests(unittest.TestCase):
    def test_frozen_containment_sources_match_pins(self):
        for name in runner.PINS:
            self.assertTrue(runner.load_frozen(name))

    def test_bad_source_pin_fails_closed(self):
        key='common-source.txt';old=runner.PINS[key]
        try:
            runner.PINS[key]='0'*64
            with self.assertRaises(RuntimeError):runner.load_frozen(key)
        finally:runner.PINS[key]=old

    def test_only_frozen_sandbox_function_is_selected(self):
        tree=ast.parse(runner.load_frozen('infer-source.txt'))
        functions=[n for n in tree.body if isinstance(n,ast.FunctionDef) and n.name=='run_sandbox']
        self.assertEqual(len(functions),1)
        text=ast.unparse(functions[0])
        for required in ['--unshare-all','--die-with-parent','--ro-bind','--seccomp','--block-fd','GGML_BACKEND_PATH']:
            self.assertIn(required,text)

    def test_frozen_sandbox_globals_including_nested_generator_are_resolved(self):
        tree=ast.parse(runner.load_frozen('infer-source.txt'))
        node=next(n for n in tree.body if isinstance(n,ast.FunctionDef) and n.name=='run_sandbox')
        required=runner.validate_sandbox_globals(node,vars(runner))
        self.assertIn('struct',required)
        missing=dict(vars(runner));missing.pop('struct')
        with self.assertRaisesRegex(RuntimeError,'struct'):runner.validate_sandbox_globals(node,missing)

    def test_invalid_scratch_rejected_before_any_native_setup(self):
        for args in [[],['/tmp/unapproved'],['/var/tmp/../tmp/unapproved'],['/var/tmp/astra-n15-05-does-not-exist']]:
            with self.subTest(args=args),self.assertRaises(RuntimeError):runner.main(args)

    def test_fixed_resource_policy(self):
        p=runner.POLICY
        self.assertEqual((p['context'],p['output_tokens'],p['threads'],p['deadline_s']),(2048,192,2,120))
        self.assertEqual(p['method'],'lexical');self.assertEqual(p['k'],1)

    def test_path_alias_requires_same_file_and_unchanged_metadata(self):
        with tempfile.TemporaryDirectory() as tmp:
            a=Path(tmp)/'a';a.write_bytes(b'fixture');b=Path(tmp)/'b';b.write_bytes(b'fixture')
            row={'source':str(a),'destination':'/qualified/fixture','sha256':'a'*64}
            approved={'native_files':[row],'notices':[],'model':row}
            actual=copy.deepcopy(approved);actual['native_files'][0]['source']=str(a.parent)+'/./a'
            runner.validate_layout(actual,approved)
            for key,value in [('source',str(b)),('sha256','b'*64),('destination','/other')]:
                bad=copy.deepcopy(actual);bad['native_files'][0][key]=value
                with self.subTest(key=key),self.assertRaises(RuntimeError):runner.validate_layout(bad,approved)

if __name__=='__main__':unittest.main()
