const CAPS={rss:536870912,storage:67108864,wall:1000,output:65536,cleanup:250};
const CALL={schema_version:'astra.n16.read-tools.v1',tool:'get_data_quality',args:{crew_id:'CREW-03',metrics:['heart_rate']}};
const CALLER={principal:{id:'n16-06g-owner-demo',role:'astronaut',crew_id:'CREW-03'},crew_ids:['CREW-03'],read_as_of:AS_OF};
const STAGED=['services/agent/designated-worker.mjs','services/agent/single-select-runtime.mjs','services/agent/single-select-contract.mjs',
  'services/agent/single-select-source.mjs','services/agent/single-quality-contract.mjs','services/agent/designated-contract.mjs','services/agent/sqlite-fixture-contract.mjs',
  'services/agent/read-contract.mjs','services/agent/containment-diagnostics.mjs','packages/contracts.mjs','docs/N16_05_SOURCE_SPEC.json'];
async function metadata(path,max=2097152){const d=await safeDirectory(dirname(path));try{
  const st=await lstat(d.at(basename(path)));check(st.isFile()&&!st.isSymbolicLink()&&st.nlink===1&&st.size<=max,'TASK_METADATA_UNSAFE');
  return await readAt(d.fd,basename(path),max);
}finally{await d.close();}}
async function atomic(name,value){const d=await safeDirectory(OUT),temp=name+'.tmp-'+randomUUID();let f;try{
  f=await open(d.at(temp),F.O_WRONLY|F.O_CREAT|F.O_EXCL|F.O_NOFOLLOW,0o600);await f.writeFile(stable(value)+'\n');await f.sync();await f.close();f=null;
  await rename(d.at(temp),d.at(name));await d.handle.sync();
}finally{if(f)await f.close();try{await unlink(d.at(temp));}catch(e){if(e.code!=='ENOENT')throw e;}await d.close();}}
async function journal(value){const f=await open(join(OUT,'resources.jsonl'),'a',0o600);try{await f.writeFile(stable({at:new Date().toISOString(),...value})+'\n');await f.sync();}finally{await f.close();}}
async function count(path){const s=await lstat(path);check(s.isDirectory()&&!s.isSymbolicLink(),'UNSAFE_ACCOUNTING_ROOT');let n=s.blocks*512;
  for(const e of await readdir(path,{withFileTypes:true})){try{const p=join(path,e.name),z=await lstat(p);n+=z.isDirectory()&&!z.isSymbolicLink()?await count(p):Math.max(z.size,z.blocks*512);}catch(e){if(e.code!=='ENOENT')throw e;}}return n;}
async function counted(){let n=0;for(const p of [DESIGNATED_ROOT,OUT,join(PROJECT,'artifacts/n16-06-20261029'),join(PROJECT,'artifacts/n16-06a-20261029'),join(PROJECT,'artifacts/n16-06f-20261029')])n+=await count(p);
  for(const p of [...STAGED,'scripts/n16-quality-qualification.mjs','services/agent/read-context.mjs','services/agent/execution-trace.mjs','services/agent/qualification-batch.mjs','services/agent/dependency-pins.mjs','tests/n16-single-quality.test.mjs','tests/n16-single-select.test.mjs']){const s=await lstat(join(PROJECT,p));n+=Math.max(s.size,s.blocks*512);}return n;}
async function tree(pid){const result=[];async function visit(id){try{const text=await readFile(`/proc/${id}/status`,'utf8'),stat=await readFile(`/proc/${id}/stat`,'utf8');
  result.push({pid:id,birth:stat.slice(stat.lastIndexOf(')')+2).split(' ')[19],hwm:Number(/^VmHWM:\s+(\d+)/m.exec(text)?.[1]??0)*1024});
  for(const child of (await readFile(`/proc/${id}/task/${id}/children`,'utf8')).trim().split(/\s+/).filter(Boolean))await visit(Number(child));
}catch(e){if(!['ENOENT','ESRCH'].includes(e.code))throw e;}}await visit(pid);return result;}
function sandbox(fd){const args=['--unshare-all','--die-with-parent','--new-session','--cap-drop','ALL','--clearenv','--setenv','PATH','/usr/bin',
  '--setenv','PYTHONDONTWRITEBYTECODE','1','--ro-bind','/usr','/usr','--ro-bind','/lib','/lib','--ro-bind','/lib64','/lib64',
  '--dir','/runtime','--ro-bind',process.execPath,'/runtime/node','--dir','/code','--ro-bind-fd',String(fd),'/fixture'];
  for(const p of STAGED)args.push('--ro-bind',join(PROJECT,p),'/code/'+p);
  args.push('--proc','/proc','--dev','/dev','--remount-ro','/','--chdir','/','/usr/bin/python3','-c',
    'import os,resource; resource.setrlimit(resource.RLIMIT_CORE,(0,0)); resource.setrlimit(resource.RLIMIT_CPU,(1,1)); os.closerange(3,256); os.execv("/runtime/node",["/runtime/node","--jitless","--max-old-space-size=64","/code/services/agent/designated-worker.mjs","n16-06f-single-select-v1"])');return args;}

export async function qualify(){
  check(process.argv.length===2,'NO_CALLER_OPTIONS');await mkdir(OUT,{recursive:true});authorizeSingle(CALL,CALLER);
  const invocation=await open(join(OUT,'invocation.json'),F.O_CREAT|F.O_EXCL|F.O_WRONLY|F.O_NOFOLLOW,0o600);
  try{await invocation.writeFile(stable({task:'N16-06G',maximum_startups:1})+'\n');await invocation.sync();}finally{await invocation.close();}
  let trace,directory,child,closed,timer,interval,escalation,reason=null,exit=null,actualStarts=0,sampleBusy=false,
    sampler=Promise.resolve(),handling=Promise.resolve(),peak=0,maxDisk=0,samples=0,started=null,ended=null,stderr='',buffer='',diagnostic=null,
    acked=false,sqlAttempts=0,eventResult=null,ownedRemaining=0,handlerError=null;
  const seen=new Map(),events=[];
  const batch=new QualificationBatch({persist:async record=>{await atomic('first-failure.json',record);return {durable:true,hash:record.hash};}});
  const stop=code=>{if(reason)return;reason=code;batch.trip({code,stage:'single_select'});
    if(child?.pid&&ended===null){try{process.kill(-child.pid,'SIGTERM');}catch(e){if(e.code!=='ESRCH')throw e;}
      escalation=setTimeout(()=>{try{process.kill(-child.pid,'SIGKILL');}catch(e){if(e.code!=='ESRCH')throw e;}},CAPS.cleanup);}};
  const sample=async()=>{const rows=child?await tree(child.pid):[];for(const p of rows)seen.set(p.pid,p.birth);
    const self=(await tree(process.pid))[0];check(self?.hwm>0,'RSS_OBSERVER_FAILED');peak=Math.max(peak,self.hwm,...rows.map(p=>p.hwm),0);
    const bytes=await counted();maxDisk=Math.max(maxDisk,bytes);samples++;await journal({phase:'resource',owned:rows,parent_hwm:self.hwm,counted_bytes:bytes});
    check(peak<CAPS.rss,'RSS_LIMIT');check(bytes<CAPS.storage,'STORAGE_LIMIT');};
  let binding,manifest,reg,spec;
  try{
    check(process.version==='v24.21.0'&&process.versions.sqlite==='3.53.4'&&process.arch==='x64'&&process.platform==='linux','RUNTIME_PIN');
    await verifyDependencyPins(CONTAINMENT_PIN_POLICY,{readInventory:p=>metadata(join(PROJECT,p)),readDependency:p=>readFile(join(PROJECT,p))});
    const plan=parseJSON((await metadata(join(OUT,'code-plan.json'))).toString(),2097152);
    for(const [p,h] of Object.entries(plan.code_pins))check(sha(await readFile(join(PROJECT,p)))===h,'CODE_PLAN_CHANGED');
    const qa=parseJSON((await metadata(join(OUT,'mandatory-qa.json'))).toString(),65536);check(qa.length===4&&qa.every(r=>r.verified===true&&r.exit_code===0),'MANDATORY_QA_FAILED');
    reg=parseJSON((await metadata(join(PROJECT,'artifacts/n16-06-20261029/registration.json'))).toString(),4096);
    check(reg.root===DESIGNATED_ROOT&&reg.manifest_sha256==='5dab2628827a80b3af1f0a5d37ef64763144aba1510e7b0c7ea653a3f5ac286c'
      &&reg.owner_sha256==='e1597530fc3236c7930990cfef4792b49782a51d781243591c0d946b645ebac1','REGISTRATION_CHANGED');
    const raw=await metadata(join(DESIGNATED_ROOT,'metadata/manifest.json'),65536);check(sha(raw)===reg.manifest_sha256,'SOURCE_MANIFEST_CHANGED');
    manifest=parseJSON(raw.toString(),65536);check(sha(await metadata(join(DESIGNATED_ROOT,'owner.json'),4096))===reg.owner_sha256,'OWNER_CHANGED');
    spec=await metadata(join(PROJECT,'docs/N16_05_SOURCE_SPEC.json'),65536);
    directory=await safeDirectory(join(DESIGNATED_ROOT,'source'));
    binding=inspectSealedSource(join(DESIGNATED_ROOT,'source'),spec,manifest.source_identity);
    check(binding.source_id===manifest.source_id&&binding.source_sha256===manifest.source_sha256,'SOURCE_MANIFEST_BINDING');
    await mkdir(join(OUT,'traces'),{recursive:true});trace=await ExecutionTrace.create({task:'N16-06G',trusted_scope:CALLER,source:binding,manifest_sha256:reg.manifest_sha256},{root:join(OUT,'traces')});
    await trace.append('CALL_REQUESTED',{call_id:'call-01'});
    await trace.append('CALL_VALIDATED',{call_id:'call-01',tool:CALL.tool,args:CALL.args,trusted_scope:CALLER});
    await sample();check(maxDisk<CAPS.storage-1048576,'STORAGE_ADMISSION_FAILED');
    await atomic('preflight.json',{status:'PASS',call:CALL,trusted_scope:CALLER,source:binding,code_pins:plan.code_pins,caps:CAPS,trace_id:trace.id,SQL_startups:0});
    await trace.append('CALL_STARTED',{call_id:'call-01',tool:CALL.tool,args:CALL.args,SQL_status:'PENDING_GUARD_ACK'});
    await batch.run('get_data_quality',async permit=>{
      const reservation=await open(join(OUT,'startup-reserved.json'),F.O_CREAT|F.O_EXCL|F.O_WRONLY|F.O_NOFOLLOW,0o600);
      try{await reservation.writeFile(stable({maximum_startups:1,trace_id:trace.id})+'\n');await reservation.sync();}finally{await reservation.close();}
      permit.markStarted();started=performance.now();child=spawn('/usr/bin/bwrap',sandbox(3),{detached:true,stdio:['pipe','pipe','pipe',directory.fd],env:{PATH:'/usr/bin'}});
      let startupRecord=Promise.resolve();child.once('spawn',()=>{actualStarts=1;startupRecord=atomic('startup.json',{actual_startups:1,pid:child.pid,trace_id:trace.id}).catch(e=>{handlerError=e;stop('STARTUP_TRACE_FAILED');});});
      permit.signal.addEventListener('abort',()=>stop('BATCH_BLOCKED'),{once:true});child.stdin.on('error',()=>{});
      interval=setInterval(()=>{if(sampleBusy)return;sampleBusy=true;sampler=sample().catch(e=>stop(e.code??'ACCOUNTING_FAILED')).finally(()=>{sampleBusy=false;});},20);
      timer=setTimeout(()=>stop('SINGLE_WORKER_TIMEOUT'),CAPS.wall);
      child.stderr.on('data',b=>{stderr+=b;if(Buffer.byteLength(stderr)>8192)stop('STDERR_LIMIT');});
      child.stdout.on('data',b=>{buffer+=b;if(Buffer.byteLength(buffer)>CAPS.output){stop('OUTPUT_LIMIT');return;}
        while(buffer.includes('\n')){const at=buffer.indexOf('\n'),line=buffer.slice(0,at);buffer=buffer.slice(at+1);
          handling=handling.then(async()=>{const e=parseJSON(line,CAPS.output);events.push(e);if(e.kernel_max_rss){peak=Math.max(peak,e.kernel_max_rss);check(peak<CAPS.rss,'RSS_LIMIT');}await journal({phase:'worker_event',event:e});
            if(e.event==='SINGLE_CONTAINMENT'){
              check(!diagnostic,'DUPLICATE_DIAGNOSTIC');diagnostic=e.diagnostic;verifySingleGuard(diagnostic);verifyBinding(e.binding,manifest.source_identity);
              check(stable(e.binding)===stable(binding)&&Object.values(e.blocked_paths).every(Boolean),'WORKER_BINDING_OR_ISOLATION');
              await startupRecord;await sample();
              await commitContainment(diagnostic,{source_id:SOURCE_ID,sha256:SOURCE_SHA},{block:f=>batch.trip(f),persist:async d=>{
                await atomic('diagnostic.json',{diagnostic:d,binding:e.binding,trace_id:trace.id});return {durable:true,hash:d.hash};},
                acknowledge:ack=>new Promise((resolve,reject)=>{batch.assertOpen();check(!reason,'WORKER_STOPPED');
                  child.stdin.write(stable({...ack,source_binding_sha256:sha(stable(binding)),call_sha256:sha(stable(CALL))})+'\n',err=>{if(err)reject(err);else{acked=true;resolve();}});})});
            }else if(e.event==='SINGLE_SQL_ATTEMPT'){check(acked&&++sqlAttempts===1,'UNAUTHORIZED_SQL_ATTEMPT');}
            else if(e.event==='SINGLE_RESULT'){check(acked&&sqlAttempts===1&&!eventResult,'INVALID_SQL_RESULT');eventResult=e;}
            else if(e.event==='FAILED')throw Object.assign(Error(e.code),{code:e.code});
            else check(false,'UNEXPECTED_WORKER_EVENT');
          }).catch(e=>{handlerError??=e;stop(e.code??'WORKER_PROTOCOL_FAILED');});
        }});
      closed=new Promise(resolve=>{child.once('error',()=>stop('WORKER_START_FAILED'));child.once('close',(code,signal)=>{ended=performance.now();resolve({code,signal});});});
      // Trace and resource persistence are established before giving the worker any request.
      await startupRecord;await sample();batch.assertOpen();check(!reason,'RESOURCE_NOT_READY');
      child.stdin.write(stable({schema_version:SINGLE_MODE,call:CALL,trusted_caller:CALLER,expected_instance:manifest.source_identity})+'\n');
      exit=await closed;clearTimeout(timer);clearTimeout(escalation);clearInterval(interval);await sampler;await handling;await startupRecord;await sample();
      for(const [pid,birth] of seen){const row=(await tree(pid))[0];if(row?.birth===birth)ownedRemaining++;}
      check(ownedRemaining===0,'WORKER_CLEANUP_FAILED');check(!handlerError,handlerError?.code??'HANDLER_FAILED');check(!reason,reason??'WORKER_FAILED');
      check(exit.code===0&&eventResult&&acked&&sqlAttempts===1,'INCOMPLETE_SQL_RESULT');
    });
    const r=eventResult.receipt,selected=expectedRows(CALL);
    check(r.mode===SINGLE_MODE&&r.readOnly&&r.explicit_statement_count===1&&r.authorizer_selects===1&&stable(r.query_ids)===stable(['observation'])
      &&r.query_sha256===sha(QUERIES.observation)&&stable(r.parameters)===stable(['CREW-03','heart_rate'])
      &&r.selected_rows_sha256===sha(stable(selected))&&r.containment_diagnostic_sha256===diagnostic.hash
      &&r.mount_namespace===diagnostic.proof.namespace,'SQL_RECEIPT_INVALID');
    check(stable(eventResult.receipt.quality_audit?.canonical_projection)===stable(projectRows('elevated',0,CALL,selected)),'QUALITY_AUDIT_CHANGED');
    check(stable(eventResult.result)===stable(assessQuality(projectRows('elevated',0,CALL,selected))),'SQL_RESULT_CHANGED');
    const after=inspectSealedSource(join(DESIGNATED_ROOT,'source'),spec,manifest.source_identity);check(stable(after)===stable(binding),'POST_READ_SOURCE_CHANGED');
    const projected=designatedProjection(eventResult.result,CALL,CALLER,reg.manifest_sha256,sha(stable(manifest.source_identity)));
    const result={run_id:trace.id,call_id:'call-01',execution_status:'COMPLETED',...projected,read_receipt:r};
    const reference=await trace.result('call-01',result);await trace.append('CALL_COMPLETED',{call_id:'call-01',tool:CALL.tool,result:reference,execution_status:'COMPLETED',data_status:result.data_status});
    await trace.append('RUN_ENDED',{status:'COMPLETED'});await trace.close();const id=trace.id;trace=null;
    const recovery=await recoverTrace(id,{root:join(OUT,'traces')});check(recovery.status==='COMPLETE'&&recovery.executed_calls.length===1&&recovery.replayed_calls===0,'RECOVERY_FAILED');
    const batchState=await batch.drain();await atomic('outcome.json',{task:'N16-06G',outcome:'PASS',actual_startups:actualStarts,sql_attempts:sqlAttempts,
      exit,diagnostic,source_before:binding,source_after:after,result,recovery,batch:batchState,events,stderr,
      native_wall_ms:ended-started,max_process_RSS:peak,max_counted_storage:maxDisk,resource_samples:samples,owned_remaining:ownedRemaining,observer_pending:sampleBusy?1:0,
      network_requests:0,downloads:0,explicit_SQL_statements:1,trace_id:id});
    return {outcome:'PASS',trace_id:id,actual_startups:actualStarts,data_status:result.data_status};
  }catch(error){stop(error.code??'QUALIFICATION_FAILED');
    if(child&&ended===null&&closed)await closed;
    clearTimeout(timer);clearTimeout(escalation);clearInterval(interval);await sampler;await handling;
    if(trace){try{await trace.append('CALL_FAILED',{call_id:'call-01',tool:CALL.tool,code:reason});await trace.append('RUN_ENDED',{status:'FAILED'});}catch{}await trace.close();}
    let state;try{state=await batch.drain();}catch{state=batch.state();}
    await atomic('outcome.json',{task:'N16-06G',outcome:'BLOCKED',reason,execution_status:'FAILED',data_status:'UNKNOWN',actual_startups:actualStarts,sql_attempts:sqlAttempts,exit,diagnostic,
      events,stderr,batch:state,native_wall_ms:ended!==null?ended-started:null,max_process_RSS:peak,max_counted_storage:maxDisk,resource_samples:samples,
      observer_pending:sampleBusy?1:0,committed_results:0});throw error;
  }finally{clearTimeout(timer);clearTimeout(escalation);clearInterval(interval);if(directory)await directory.close();}
}
if(process.argv[1]&&resolve(process.argv[1])===fileURLToPath(import.meta.url)){
  try{process.stdout.write(JSON.stringify(await qualify())+'\n');}catch(e){process.stderr.write(JSON.stringify({outcome:'BLOCKED',code:e.code??'PREFLIGHT_FAILED'})+'\n');process.exitCode=1;}
}
